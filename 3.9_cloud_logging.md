# 3.9 การบันทึกข้อมูลลง Cloud

หัวข้อนี้อธิบายการบันทึกข้อมูลลง Cloud ของระบบ โดยอ้างอิงจากโค้ดจริงในโปรเจกต์เท่านั้น ระบบมีการส่งข้อมูลขึ้น Supabase 2 กลุ่มหลัก ได้แก่

1. ข้อมูลผลการตรวจสอบชิ้นงาน ไปยังตาราง `part_records`
2. ข้อมูลสถานะเครื่องและระบบ ไปยังตาราง `system_status`

## 3.9.1 การตั้งค่าการเชื่อมต่อ Supabase

ระบบอ่านค่าการเชื่อมต่อ Supabase จากไฟล์ `config.json` หรือ environment variable โดยโค้ดอยู่ในไฟล์ `core/project_config.py`

```python
BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
CONFIG_PATH = os.path.join(BASE_DIR, "config.json")
DEFAULT_SUPABASE_URL = "https://ptxfbxwufbrivfrcplku.supabase.co"
DEFAULT_SUPABASE_KEY = ""
```

โค้ดส่วนนี้กำหนดตำแหน่งไฟล์ `config.json` และค่า default สำหรับ Supabase โดย `DEFAULT_SUPABASE_KEY` ถูกตั้งเป็นค่าว่าง เพื่อไม่ให้ API Key จริงถูกเปิดเผยในโค้ด

```python
def load_local_config():
    if not os.path.exists(CONFIG_PATH):
        return {}
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as fh:
            data = json.load(fh)
            return data if isinstance(data, dict) else {}
    except Exception:
        return {}
```

ฟังก์ชัน `load_local_config()` ใช้อ่านไฟล์ `config.json` ถ้าไม่มีไฟล์หรืออ่านไม่ได้จะคืนค่าเป็น `{}` เพื่อให้ระบบยังทำงานต่อได้โดยใช้ค่า default หรือ environment variable

```python
def get_supabase_settings():
    cfg = load_local_config()
    base_url = str(os.getenv("SUPABASE_URL") or cfg.get("SUPABASE_URL") or DEFAULT_SUPABASE_URL).rstrip("/")
    api_key = str(os.getenv("SUPABASE_KEY") or cfg.get("SUPABASE_KEY") or DEFAULT_SUPABASE_KEY).strip()
    return base_url, api_key
```

ฟังก์ชัน `get_supabase_settings()` ใช้อ่านค่า `SUPABASE_URL` และ `SUPABASE_KEY` โดยเรียงลำดับจาก environment variable ก่อน ถ้าไม่มีจึงอ่านจาก `config.json` และถ้ายังไม่มีจะใช้ค่า default

ตัวอย่างค่าในไฟล์ `config.example.json`

```json
{
  "SUPABASE_URL": "https://your-project-ref.supabase.co",
  "SUPABASE_KEY": "replace-with-supabase-anon-or-service-key"
}
```

ดังนั้น `SUPABASE_KEY` คือ API Key ที่ระบบใช้ยืนยันสิทธิ์ในการเรียก Supabase REST API

## 3.9.2 ฟังก์ชันสำหรับส่งข้อมูลแบบ JSON

ระบบใช้ helper ในไฟล์ `services/http_utils.py` สำหรับส่งข้อมูล JSON ไปยัง Supabase

```python
def post_json_with_retry(url, payload, headers, timeout=(3.0, 10.0)):
    return http_session.post(url, json=payload, headers=headers, timeout=timeout)
```

ฟังก์ชันนี้ใช้ส่งข้อมูลแบบ `POST` โดยส่ง `payload` เป็น JSON พร้อม `headers` ที่มี API Key และ Authorization

```python
def parse_missing_schema_column(response_text):
    text = str(response_text or "")
    match = re.search(r"Could not find the '([^']+)' column", text)
    return match.group(1) if match else None
```

ฟังก์ชันนี้ใช้ตรวจข้อความ error จาก Supabase ว่ามีคอลัมน์ใดใน payload ที่ไม่มีอยู่ใน schema ของตาราง

```python
def post_json_pruning_unknown_columns(url, payload, headers, timeout=(3.0, 10.0), max_attempts=None):
    working_payload = dict(payload or {})
    removed_columns = []
    attempts_left = max_attempts or max(1, len(working_payload))
    last_response = None

    while attempts_left > 0 and working_payload:
        last_response = post_json_with_retry(url, working_payload, headers, timeout=timeout)
        if last_response.status_code in (200, 201):
            return last_response, removed_columns, working_payload

        if last_response.status_code != 400:
            return last_response, removed_columns, working_payload

        missing_column = parse_missing_schema_column(last_response.text)
        if not missing_column or missing_column not in working_payload:
            return last_response, removed_columns, working_payload

        removed_columns.append(missing_column)
        working_payload.pop(missing_column, None)
        attempts_left -= 1

    return last_response, removed_columns, working_payload
```

ฟังก์ชัน `post_json_pruning_unknown_columns()` เป็นฟังก์ชันส่งข้อมูลหลักที่ใช้ในระบบ หาก Supabase ตอบกลับว่า payload มีคอลัมน์ที่ไม่มีในตาราง ฟังก์ชันนี้จะลบคอลัมน์นั้นออกแล้วลองส่งใหม่ ทำให้ระบบยังสามารถบันทึกข้อมูลที่เหลือได้

## 3.9.3 การส่งข้อมูลผลตรวจชิ้นงาน

เมื่อระบบตรวจครบทุกด้าน ฟังก์ชัน `complete_part_if_ready()` ในไฟล์ `app.py` จะเรียกบันทึกผลตรวจลง Supabase

```python
def complete_part_if_ready():
    global part_session, locked_overlay, latest_ai_results, inspection_active, overlay_block_until
    global side3_measurement_armed, side3_manual_preview_enabled
    final_session = None
    final_result = None

    with data_lock:
        sides_complete = all(part_session.get(f"side{i}") for i in ACTIVE_INSPECTION_SIDES)
        if not sides_complete:
            return None

        final_session = part_session.copy()
        final_result = compute_final_result(final_session)
        part_session = reset_part_session()
        reset_dimension_state(part_session)
        reset_side_observation(1)
        reset_side3_smoothing_state()
        side3_measurement_armed = False
        side3_manual_preview_enabled = False
        locked_overlay = None
        latest_ai_results = default_ai_results(1)
        inspection_active = False
        overlay_block_until = time.time() + SIDE_SWITCH_OVERLAY_BLOCK_SEC

    trigger_modbus_signal(4 if final_result == "NG" else 3)
    log_part_to_supabase(final_session)
    return {
        "session": final_session,
        "final_result": final_result,
    }
```

โค้ดนี้ตรวจว่า `side1`, `side2` และ `side3` มีผลครบแล้วหรือยัง ถ้าครบแล้วจะคำนวณผลรวมด้วย `compute_final_result()` จากนั้นเรียก `log_part_to_supabase(final_session)` เพื่อส่งข้อมูลผลตรวจขึ้น Cloud

### การสร้าง Payload สำหรับตาราง `part_records`

โค้ดสร้าง payload อยู่ในฟังก์ชัน `_build_part_record_payload()` ของไฟล์ `app.py`

```python
def _build_part_record_payload(session, curr_id, ts, res, defect_s1, defect_s2, defect_s3):
    return {
        "part_id": curr_id,
        "record_timestamp": ts,
        "side1": session["side1"],
        "side2": session["side2"],
        "side3": session["side3"],
        "defect _s1": defect_s1,
        "defect _s2": defect_s2,
        "defect _s3": defect_s3,
        "capture_s1": session.get("capture_s1"),
        "capture_s2": session.get("capture_s2"),
        "capture_s3": session.get("capture_s3"),
        "dimension of top": session.get("dimension_top"),
        "dimension of bottom": session.get("dimension_bottom"),
        "dimension of length": session.get("dimension_length"),
        "result": res,
    }
```

payload นี้คือข้อมูลที่จะถูกส่งไปยังตาราง `part_records` โดยประกอบด้วยหมายเลขชิ้นงาน เวลา ผลตรวจแต่ละด้าน defect ของแต่ละด้าน URL รูปภาพของแต่ละด้าน ค่าการวัดขนาดของ SIDE 3 และผลรวมของชิ้นงาน

### การส่งข้อมูลไปยัง Supabase

ฟังก์ชัน `log_part_to_supabase()` ในไฟล์ `app.py` เป็นฟังก์ชันหลักสำหรับบันทึกผลตรวจชิ้นงาน

```python
def log_part_to_supabase(session):
    global total_parts_inspected
    defect_s1 = "-" if str(session.get("side1", "")).startswith("GOOD") else session.get("defect_s1")
    defect_s2 = "-" if str(session.get("side2", "")).startswith("GOOD") else session.get("defect_s2")
    defect_s3 = "-" if str(session.get("side3", "")).startswith("GOOD") else session.get("defect_s3")
    curr_id = int(session.get("part_id") or 0)
    if curr_id <= 0:
        curr_id = reserve_next_part_id()
```

ส่วนนี้เตรียมข้อมูล defect ของแต่ละด้าน หากด้านใดเป็น `GOOD` จะกำหนด defect เป็น `"-"` แต่ถ้าเป็น `NG` จะใช้ defect ที่ตรวจพบ และถ้ายังไม่มี `part_id` ระบบจะจองเลขชิ้นงานใหม่

```python
    with data_lock:

        entry = {
            "timestamp": ts,

            "side2": session["side2"],
            "side3": session["side3"],
            "s1": session["side1"],
            "s2": session["side2"],
            "s3": session["side3"],
            "defect_s1": defect_s1,
            "defect_s2": defect_s2,
            "defect_s3": defect_s3,
            "capture_s1": session.get("capture_s1"),
            "capture_s2": session.get("capture_s2"),
            "capture_s3": session.get("capture_s3"),
            "dimension_top": session.get("dimension_top"),
            "dimension_bottom": session.get("dimension_bottom"),
            "dimension_length": session.get("dimension_length"),
            "dimension_status": session.get("dimension_status"),
        }
        inspection_history.append(entry)
        if len(inspection_history) > 100: inspection_history.pop(0)
```

ส่วนนี้สร้างเวลาบันทึกข้อมูล คำนวณผลรวมของชิ้นงาน และเก็บข้อมูลไว้ใน `inspection_history` เพื่อให้หน้าเว็บสามารถแสดงประวัติการตรวจสอบล่าสุดได้

```python
    # Supabase Connection
    base_url, key = get_supabase_settings()
    url = f"{base_url}/rest/v1/part_records"
    headers = {"apikey": key, "Authorization": f"Bearer {key}", "Content-Type": "application/json"}
    payload = _build_part_record_payload(session, curr_id, ts, res, defect_s1, defect_s2, defect_s3)
    try:
        response, removed_columns, sent_payload = post_json_pruning_unknown_columns(
            url,
            payload,
            headers,
            timeout=(3.0, 10.0),
        )
        if response.status_code in (200, 201):
            removed_note = f", removed={','.join(removed_columns)}" if removed_columns else ""
            print(
                f"[LOG] Part recorded. Status: {response.status_code}, keys_sent={len(sent_payload)}{removed_note}",
                flush=True,
            )
            if str(res).upper() == "NG":
                send_ng_alert_async(session.copy(), curr_id, ts, defect_s1, defect_s2, defect_s3)
            return

        print(f"[LOG] Part record failed: {response.status_code} - {response.text}", flush=True)
    except Exception as e:
        print(f"[LOG] Error: {e}", flush=True)
```

ส่วนนี้เป็นการเชื่อมต่อ Supabase โดยใช้ `get_supabase_settings()` เพื่อดึง URL และ API Key จากนั้นสร้าง endpoint ไปยังตาราง `part_records` และส่ง payload ด้วย `post_json_pruning_unknown_columns()` หากบันทึกสำเร็จ Supabase จะตอบกลับ status code `200` หรือ `201` และถ้าผลรวมเป็น `NG` ระบบจะเรียก `send_ng_alert_async()` เพื่อแจ้งเตือนเพิ่มเติม

## 3.9.4 การส่งข้อมูลสถานะเครื่องและระบบ

นอกจากข้อมูลผลตรวจชิ้นงาน ระบบยังส่งข้อมูลสถานะเครื่องขึ้น Supabase ด้วย โดยโค้ดอยู่ในฟังก์ชัน `system_monitor_thread()` ของไฟล์ `app.py`

```python
def system_monitor_thread():
    while True:
        try:
            base_url, key = get_supabase_settings()
            url = f"{base_url}/rest/v1/system_status"
            headers = {"apikey": key, "Authorization": f"Bearer {key}", "Content-Type": "application/json"}
            with data_lock:
                ai_res = latest_ai_results.copy()
            stats = get_pi_stats()
            printer = get_printer_status(force_refresh=False)
            all_ips = get_all_ips()
            nozzle_temp = _as_float(printer.get("nozzle_temp"), 0.0)
            bed_temp = _as_float(printer.get("bed_temp"), 0.0)
```

ส่วนนี้เชื่อมต่อ Supabase แล้วกำหนด endpoint เป็นตาราง `system_status` จากนั้นอ่านข้อมูลสถานะของ Raspberry Pi, Printer, IP และอุณหภูมิของหัวฉีดกับฐานพิมพ์

```python
            base_payload = {
                "pi_cpu_usage": stats.get("cpu_usage"),
                "pi_ram_usage": stats.get("ram_usage"),
                "pi_disk_usage": stats.get("disk_usage"),
                "pi_cpu_temp": stats.get("cpu_temp"),
                "robot_status": get_robot_status(),
                "system_active": system_on,
                "printer_status": str(printer.get("status") or "Disconnected"),
                "printer_nozzle_temp": nozzle_temp,
                "printer_bed_temp": bed_temp,
                "server_ip": all_ips[0] if len(all_ips) > 0 else "127.0.0.1",
                "modbus_port": 5020,
                "timestamp": datetime.datetime.now(UTC).isoformat(),
            }
```

`base_payload` คือข้อมูลสถานะหลักของระบบ ประกอบด้วย CPU, RAM, Disk, อุณหภูมิ CPU ของ Raspberry Pi, สถานะ Robot, สถานะเปิด/ปิดระบบ, สถานะ Printer, IP ของ Server, Modbus Port และเวลา

```python
            extended_payload = {
                **base_payload,
                "printer_progress": _as_float(printer.get("percent"), 0.0),
                "printer_task_name": str(printer.get("task_name") or "").strip(),
                "printer_remaining_time": _as_int(printer.get("remaining_time"), 0),
                "printer_sub_stage": str(printer.get("sub_stage") or "").strip(),
            }
```

`extended_payload` เพิ่มข้อมูลรายละเอียดของ Printer เช่น เปอร์เซ็นต์งานพิมพ์ ชื่องานพิมพ์ เวลาที่เหลือ และสถานะย่อยของ Printer

```python
            response, removed_columns, sent_payload = post_json_pruning_unknown_columns(
                url,
                extended_payload,
                headers,
                timeout=(3.0, 10.0),
            )
            if response.status_code not in (200, 201):
                log_system_status_sync(f"status={response.status_code}, body={response.text[:240]}")
            else:
                removed_note = ""
                if removed_columns:
                    removed_note = f", removed={','.join(removed_columns)}"
                log_system_status_sync(
                    "ok "
                    f"printer_status={extended_payload['printer_status']}, "
                    f"progress={extended_payload['printer_progress']}, "
                    f"task={extended_payload['printer_task_name'] or '-'}"
                    f"{removed_note}, keys_sent={len(sent_payload)}"
                )
        except Exception as e:
            log_system_status_sync(f"exception: {e}")
        time.sleep(30)
```

ส่วนนี้ส่งข้อมูลสถานะเครื่องไปยัง Supabase ตาราง `system_status` ถ้าส่งไม่สำเร็จจะบันทึกข้อความ error แต่ถ้าสำเร็จจะ log สถานะการส่งข้อมูล โดยฟังก์ชันนี้ทำงานซ้ำทุก 30 วินาทีจาก `time.sleep(30)`

## 3.9.5 สรุปการทำงาน

ลำดับการส่งข้อมูลของระบบจริงสามารถสรุปได้ดังนี้

```text
ตรวจชิ้นงานครบ SIDE 1, SIDE 2, SIDE 3
→ คำนวณผลรวม GOOD/NG
→ สร้าง payload ผลตรวจชิ้นงาน
→ ส่งไป Supabase ตาราง part_records

พร้อมกันนั้น system_monitor_thread ทำงานวนซ้ำ
→ อ่านสถานะ Raspberry Pi, Robot, Printer, IP, Modbus
→ สร้าง payload สถานะระบบ
→ ส่งไป Supabase ตาราง system_status
→ รอ 30 วินาทีแล้วส่งซ้ำ
```

ดังนั้นระบบนี้ไม่ได้บันทึกเฉพาะผลตรวจของชิ้นงานเท่านั้น แต่ยังบันทึกสถานะการทำงานของเครื่องและอุปกรณ์ที่เกี่ยวข้องขึ้น Cloud เพื่อใช้ติดตามการทำงานของระบบแบบต่อเนื่อง
